\begin{itemize}

  \item \emph{Hogy vagy?} `How are you?' inform. --- \emph{Hogy van?}
    formal.

  \item \emph{Megvayok.} `I am all right.' (neutral)

  \item \emph{Rosszul vagyok.} `I am ill/sick.' (\emph{rossz} `bad')

  \item \emph{ön}: pers. pron. `you' sg. form.; \emph{önök} `you'
    pl. form.

  \item \emph{Mi újság?} `What's up?' inform.

  \item \emph{Is veled?} `And what with you?' (\emph{te} `you'
    sg. inform. + \emph{-vel} `with' $\rightarrow$ \emph{veled})

  \item \emph{Semmi különös/extra.} `Nothing special.'

  \item \emph{Minden rendben.} `Everything is fine.'

  \item \emph{Milyen idő van?} or \emph{Milyen az idő?} `What's the
    weather?'

  \item \emph{Hideg van.} `It is cold.' or \emph{Hideg idő van.} `The
    weather is cold.'

  \item \emph{Jól/Rossz idő van.} `The weather is good/bad.'

  \item \emph{Szeles idő van} `The weather is windy.' --- \emph{szél}
    `wind'

  \item \emph{Fúj a szél.} `The wind is blowing.'

  \item \emph{Eszik az eső.} `The rain is falling.'; \emph{Eszik a
  hó.} `The snow is falling.'; \emph{Esős idő van.} `The weather is
    rainy.'

  \item \emph{Fázom.} `I am/feel cold.' (vb. ind. \emph{fázik} `to be
    cold').

  \item \emph{Süt a nap.} `The sun shines.'; \emph{Napos idő van.}
    `The weather is sunny.'

  \item \emph{Felhós idő van.} `The weather is cloudy.'

  \item \emph{Hűvös idő van.} `The weather is chilly.'

  \item \emph{Hány fok van?} `How many degrees it is?'

\end{itemize}

Example of small talk:

\begin{itemize}

  \item[---] \emph{Szia! Hogy vagy?}

  \item[---] \emph{Szia! Jól vagyok, csak egy kicsit fáradt vagyok. És
  te?}

  \item[---] \emph{Megvagyok, köszönöm. Mi újság?}

  \item[---] \emph{Semmi különös. Is veled?}

  \item[---] \emph{Is semmi extra. Egerben milyen idő van?}

  \item[---] \emph{Itt hűvós az idő. Budapesten hány fok van?}

\end{itemize}
